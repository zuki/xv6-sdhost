#ifndef INC_SDIO_H
#define INC_SDIO_H

#include <types.h>
#include <sdio.h>

struct tscr // SD configuration register
{
    uint32_t    scr[2];
    uint32_t    sd_bus_widths;
    int    sd_version;
};

struct sdio {
    uint64_t ull_offset;
    uint32_t hci_ver;

    // was: struct sdio_block_dev
    uint32_t device_id[4];

    uint32_t supports_uhsi;
    uint32_t supports_18v;
    uint32_t supports_4bit;
    uint32_t ocr;
    uint32_t rca;
    uint32_t bus_speed;

    uint32_t last_interrupt;
    uint32_t last_error;

    struct tscr scr;

    int failed_voltage_switch;

    uint32_t last_cmd_reg;
    uint32_t last_cmd;
    uint32_t last_cmd_success;
    uint32_t last_r0;
    uint32_t last_r1;
    uint32_t last_r2;
    uint32_t last_r3;

    void *buf;
    int blocks_to_transfer;
    size_t block_size;
    int card_removal;
    uint32_t base_clock;

};

void sdio_init(void);
void sdio_reinit(void);
void sdio_deinit(void);
void sdio_set_irq(bool enable);
void sdio_enable_high_speed_4bit(void);
int sdio_transfer(uint32_t cmd, uint32_t arg, uint32_t *resp);
int sdio_transfer_cmd53(bool write, uint32_t block_size, uint32_t arg, size_t len, uint8_t *buf);
